tO = 3;     % outside temperature
Q  = 5;     % heater power

tW0 = 4;    % wall temperature

State-space model

A = [-1/(Rr*C1),  1/(Rr*C1);
      1/(Rr*C2), -(1/Rr + 1/Rw)/C2];
B = [1/C1, 0;
     0,    1/(Rw*C2)];
sys = ss(A, B, eye(2), zeros(2)); % eye is identity matrix

Simulate
